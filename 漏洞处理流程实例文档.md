```python
class CVEExtractor(SourceExtractor):
    def _extract_with_rules(self, content: str) -> Dict:
        soup = BeautifulSoup(content, 'html.parser')
        data = {}
        
        # CVE ID 提取
        h1_tag = soup.find('h1')
        if h1_tag:
            cve_match = re.search(r'CVE-\d{4}-\d{4,7}', h1_tag.get_text())
            if cve_match:
                data['cve_id'] = cve_match.group()
        
        # 标题提取
        title_tag = soup.find('title')
        if title_tag:
            data['title'] = title_tag.get_text().strip()
        
        # 描述提取
        desc_div = soup.find('div', class_='description')
        if desc_div:
            desc_p = desc_div.find('p')
            if desc_p:
                data['description'] = desc_p.get_text().strip()
        
        # 严重程度提取
        severity_p = soup.find('p', class_='severity')
        if severity_p:
            data['severity'] = severity_p.get_text().strip()
        
        # CVSS评分提取
        cvss_text = soup.get_text()
        cvss_score_match = re.search(r'Base Score:\s*(\d+\.\d+)', cvss_text)
        if cvss_score_match:
            data['cvss3_base_score'] = float(cvss_score_match.group(1))
        
        cvss_vector_match = re.search(r'Vector:\s*(CVSS:3\.1/[^\s]+)', cvss_text)
        if cvss_vector_match:
            data['cvss3_vector'] = cvss_vector_match.group(1)
        
        # 日期提取
        published_match = re.search(r'Published Date[:\s]*(\d{4}-\d{2}-\d{2})', cvss_text)
        if published_match:
            data['published_date'] = published_match.group(1)
        
        return data
```

```python
class DataCleaner:
    def clean(self, data: Dict) -> Dict:
        cleaned_data = {}
        
        for key, value in data.items():
            if isinstance(value, str):
                # HTML标签清理
                value = self._remove_html_tags(value)

                value = self._normalize_whitespace(value)
```

```python
def validate(self, data: Dict, source: str) -> Dict:
    validation_result = {
        'is_valid': True,
        'errors': [],
        'warnings': [],
        'field_scores': {}
    }
    
    field_config = self.field_config_loader.get_source_config(source)
    
    for field_name, field_info in field_config['fields'].items():
        value = data.get(field_name)
        field_score = 1.0
        
        # 必需字段检查
        if field_info.get('required', False) and not value:
            validation_result['errors'].append(f"缺少必需字段: {field_name}")
            field_score = 0.0
        
        # 数据类型检查
        if value is not None:
            expected_type = field_info.get('type', 'str')
            if not self._validate_type(value, expected_type):
                validation_result['warnings'].append(f"字段类型不匹配: {field_name}")
                field_score *= 0.8
        
        # CVE ID格式验证
        if field_name == 'cve_id' and value:
            if not re.match(r'^CVE-\d{4}-\d{4,7}$', value):
                validation_result['errors'].append(f"CVE ID格式错误: {value}")
                field_score = 0.0
        
        # 日期格式验证
        if 'date' in field_name and value:
            try:
                datetime.strptime(value, '%Y-%m-%d')
            except ValueError:
                validation_result['warnings'].append(f"日期格式可能有误: {field_name}")
                field_score *= 0.9
        
        validation_result['field_scores'][field_name] = field_score
    
    # 整体验证结果
    if validation_result['errors']:
        validation_result['is_valid'] = False
    
    return validation_result
```

#### 3.3 质量评分
```python
def calculate_quality_score(self, data: Dict, validation_result: Dict, 
                          extraction_confidence: float) -> float:
    # 基础分数：字段完整性
    field_scores = validation_result.get('field_scores', {})
    completeness_score = sum(field_scores.values()) / len(field_scores) if field_scores else 0
    
    # 准确性分数：基于验证结果
    accuracy_score = 1.0
    if validation_result.get('errors'):
        accuracy_score -= 0.3 * len(validation_result['errors'])
    if validation_result.get('warnings'):
        accuracy_score -= 0.1 * len(validation_result['warnings'])
    
    # 一致性分数：字段间逻辑一致性
    consistency_score = self._check_consistency(data)
    
    # 综合质量分数
    quality_score = (
        completeness_score * 0.4 +
        accuracy_score * 0.3 +
        consistency_score * 0.2 +
        extraction_confidence * 0.1
    )
    
    return min(max(quality_score, 0.0), 1.0)
```

#### 5.2 主表数据存储
```python
def upsert(self, data: Dict, source: str) -> bool:
    model_class = self.get_model(source)
    
    # 数据类型验证和转换
    validated_data = self._validate_data(data, source)
    
    # 查找现有记录
    existing = self.session.query(model_class).filter(
        model_class.cve_id == validated_data['cve_id']
    ).first()
    
    if existing:
        # 更新现有记录
        for key, value in validated_data.items():
            setattr(existing, key, value)
        existing.updated_at = datetime.utcnow()
        logger.info(f"更新CVE记录: {validated_data['cve_id']}")
    else:
        # 插入新记录
        new_record = model_class(**validated_data)
        self.session.add(new_record)
        logger.info(f"插入新CVE记录: {validated_data['cve_id']}")
    
    self.session.commit()
    return True
```
